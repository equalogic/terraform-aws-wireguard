#cloud-config
package_update: true
package_upgrade: true

write_files:
  - path: /etc/wireguard/wg0.conf
    owner: root:root
    permissions: '0600'
    content: |
      [Interface]
      Address = ${wg_server_net}
      PrivateKey = ${wg_server_private_key}
      ListenPort = ${wg_server_port}
      PostUp   = iptables -A FORWARD -i %i -j ACCEPT; iptables -A FORWARD -o %i -j ACCEPT; iptables -t nat -A POSTROUTING -o ${wg_server_interface} -j MASQUERADE
      PostDown = iptables -D FORWARD -i %i -j ACCEPT; iptables -D FORWARD -o %i -j ACCEPT; iptables -t nat -D POSTROUTING -o ${wg_server_interface}

      ${peers}

  - path: /etc/wireguard/wg-bootstrap.sh
    owner: root:root
    permissions: '0700'
    content: |
      #!/bin/bash
      set -eux

      # Install packages based on Ubuntu version
      RELEASE=$(lsb_release -rs)
      if [ "$RELEASE" = "20.04" ]; then
        apt-get install -y wireguard-dkms wireguard-tools awscli
      else
        apt-get install -y wireguard-tools awscli2
      fi

      # Associate Elastic IP if needed
      if [ "${use_eip}" != "disabled" ]; then
        INSTANCE_ID=$(curl -s http://169.254.169.254/latest/meta-data/instance-id)
        REGION=$(curl -fsq http://169.254.169.254/latest/meta-data/placement/availability-zone | sed 's/[a-z]$//')
        aws --region $REGION ec2 associate-address --allocation-id ${eip_id} --instance-id $INSTANCE_ID
      fi

      # Networking, firewall, and WireGuard service
      ip link set dev ${wg_server_interface} mtu 1420
      sed -i 's/#net.ipv4.ip_forward=1/net.ipv4.ip_forward=1/' /etc/sysctl.conf
      sysctl -p
      ufw allow ssh
      ufw allow ${wg_server_port}/udp
      ufw --force enable
      systemctl enable wg-quick@wg0.service
      systemctl start wg-quick@wg0.service

runcmd:
  - /etc/wireguard/wg-bootstrap.sh
